/* ============ EDIT YOUR FIGURES HERE ============
   a = amount collected so far, in rupees (digits only, no commas)
   g = target in rupees for that batch, or 0 for an open target
   updated = the text shown as "Figures updated ..." at the bottom
   howTo = the donation details shown under "How to give"
   Save the file after editing, then upload it again to your host. */
var DATA = {
  school: "Jawahar Navodaya Vidyalaya Bagalkot",
  tagline: "Eighteen batches, one cup. Every donation moves your batch's meter.",
  howTo: "96901001495kagb@cnrb",
  updated: "5 Nov 2026, 1:27 pm",
  sample: false,
  batches: [
    {n:"1st Batch", a:69000, g:200000},
    {n:"2nd Batch", a:0, g:200000},
    {n:"3rd Batch", a:0, g:200000},

    {n:"13th Batch", a:0, g:0},
    {n:"14th Batch", a:0, g:0},
    {n:"15th Batch", a:0, g:0},

    {n:"18th Batch", a:0, g:0}
  ]
};

<script>

(function(){
  var S = DATA;
  var sortMode = 'batch';
  function esc(s){ return String(s).replace(/[&<>"']/g, function(c){ return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]; }); }
  function inr(n){ return '₹' + Math.round(n).toLocaleString('en-IN'); }
  function tier(b){ return b.g > 0 ? (b.p >= 100 ? 'done' : b.p >= 70 ? 'close' : 'build') : 'open'; }
  function parseNum(v){ return Math.max(0, parseInt(String(v).replace(/[^\d]/g,''), 10) || 0); }
  function fmtDate(iso){
    try { return new Date(iso).toLocaleString('en-IN', {day:'numeric', month:'short', year:'numeric', hour:'numeric', minute:'2-digit'}); }
    catch(e){ return ''; }
  }

  function compute(){
    var list = S.batches.map(function(b, i){ return {i:i, n:b.n, a:b.a, g:b.g || 0}; });
    var maxG = list.reduce(function(m, b){ return Math.max(m, b.g); }, 1);
    list.forEach(function(b){ b.p = b.g > 0 ? b.a / b.g * 100 : b.a / maxG * 100; });
    var ranked = list.slice().sort(function(x, y){ return (y.a - x.a) || (x.i - y.i); });
    ranked.forEach(function(b, k){
      b.rank = (k > 0 && b.a === ranked[k-1].a) ? ranked[k-1].rank : k + 1;
      b.ahead = null;
      for (var j = k - 1; j >= 0; j--) { if (ranked[j].a > b.a) { b.ahead = ranked[j]; break; } }
    });
    var fixed = list.filter(function(b){ return b.g > 0; });
    return {
      list:list, ranked:ranked, fixed:fixed,
      total: list.reduce(function(s, b){ return s + b.a; }, 0),
      targetSum: fixed.reduce(function(s, b){ return s + b.g; }, 0),
      fixedRaised: fixed.reduce(function(s, b){ return s + b.a; }, 0),
      nOpen: list.length - fixed.length
    };
  }

  function gapLine(b, c){
    if (b.g > 0 && b.p >= 100) return '<div class="gap done">Goal reached</div>';
    if (b.a === 0 && b.rank === 1) return '<div class="gap">Nobody has given yet. Be the first.</div>';
    if (b.rank === 1) {
      var second = c.ranked.filter(function(x){ return x.a < b.a; })[0];
      return '<div class="gap lead">Leading' + (second ? ' by ' + inr(b.a - second.a) : '') + '</div>';
    }
    if (b.ahead) return '<div class="gap">' + inr(b.ahead.a - b.a) + ' to catch ' + esc(b.ahead.n) + '</div>';
    return '';
  }

  function jarHtml(b, c){
    var p = Math.min(100, b.p);
    var first = b.rank === 1 && b.a > 0;
    return '<li class="line t-' + tier(b) + '">' +
      '<div class="l-name"><span>' + esc(b.n) + '</span><b class="l-rank' + (first ? ' first' : '') + '">Rank ' + b.rank + '</b></div>' +
      '<div class="l-amt">' + inr(b.a) + '</div>' +
      '<div class="l-bar"><div class="l-fill' + (b.a === 0 ? ' empty' : '') + '" style="--p:' + p.toFixed(1) + '"></div></div>' +
      '<div class="l-sub"><span class="pct">' + (b.g > 0 ? Math.round(b.p) + '% of ' + inr(b.g) : 'Open target') + '</span>' + gapLine(b, c) + '</div>' +
    '</li>';
  }

  function boardHtml(c){
    var items = sortMode === 'rank' ? c.ranked : c.list;
    return items.map(function(b){ return jarHtml(b, c); }).join('');
  }

  function render(){
    var c = compute();
    var tp = c.targetSum > 0 ? c.fixedRaised / c.targetSum * 100 : 0;
    var atGoal = c.fixed.filter(function(b){ return b.p >= 100; }).length;
    var lead = c.ranked[0];
    var togo = Math.max(0, c.targetSum - c.fixedRaised);
    var scale = [0, 25, 50, 75, 100].map(function(t){
      var label = t === 0 ? '₹0' : t === 100 ? inr(c.targetSum) : t + '%';
      return '<span style="left:' + t + '%">' + label + '</span>';
    }).join('');

    document.getElementById('app').innerHTML =
    '<div class="wrap">' +
      (S.sample ? '<div class="banner"><b>Sample figures</b><span>' + 'These are placeholders until the organiser enters the real amounts.' + '</span></div>' : '') +
      '<section class="hero" aria-label="Total raised">' +
        '<div class="hero-top"><div><p class="eyebrow">' + esc(S.school) + ' · Alumni fund drive</p>' +
        '<h1>The Batch Cup</h1><p class="tag">' + esc(S.tagline) + '</p></div>' +
        '' + '</div>' +
        '<div class="total"><span class="big">' + inr(c.total) + '</span><span class="of">raised in total</span></div>' +
        '<div><div class="rail" role="img" aria-label="' + Math.round(tp) + ' percent of the target raised"><div class="rail-fill" style="--p:' + Math.min(100, tp).toFixed(1) + '"></div></div><div class="scale">' + scale + '</div></div>' +
        '<p class="railnote">Batches with a target: ' + inr(c.fixedRaised) + ' of ' + inr(c.targetSum) + ' (' + Math.round(tp) + '%)' + (c.nOpen ? '. ' + c.nOpen + ' open-target batches count toward the total only.' : '') + '</p>' +
        '<dl class="stats">' +
          '<div><dt>Batches at goal</dt><dd class="num">' + atGoal + ' of ' + c.fixed.length + '</dd></div>' +
          '<div><dt>Leading</dt><dd>' + (lead && lead.a > 0 ? esc(lead.n) : 'No one yet') + '</dd></div>' +
          '<div><dt>Still to raise</dt><dd class="num">' + inr(togo) + '</dd></div>' +
        '</dl>' +
      '</section>' +
      '<section aria-label="Funds by batch">' +
        '<div class="board-head"><div><h2>Batch by batch</h2><p class="sub">One line per batch, filled toward its own target. Open-target batches fill against the largest target.</p></div>' +
        '<div class="chips" role="group" aria-label="Order"><button class="chip" data-act="sort" data-mode="batch" aria-pressed="' + (sortMode==='batch') + '" type="button">By batch</button><button class="chip" data-act="sort" data-mode="rank" aria-pressed="' + (sortMode==='rank') + '" type="button">By rank</button></div></div>' +
        '<ol class="board" id="board" style="margin-top:20px">' + boardHtml(c) + '</ol>' +
      '</section>' +
      '<section class="give" aria-label="How to give"><h2>How to give</h2><p class="howto" id="howto">' + esc(S.howTo) + '</p>' +
        '<div class="row-actions"><button class="btn" data-act="copy" type="button">Copy</button></div></section>' +
      '<p class="foot">Figures updated ' + esc(S.updated) + '</p>' +
    '</div>';
  }

  
  async function copyHow(btn){
    var text = S.howTo;
    try { await navigator.clipboard.writeText(text); btn.textContent = 'Copied'; }
    catch(e){
      var r = document.createRange(); r.selectNodeContents(document.getElementById('howto'));
      var sel = window.getSelection(); sel.removeAllRanges(); sel.addRange(r);
      btn.textContent = 'Selected';
    }
    setTimeout(function(){ btn.textContent = 'Copy'; }, 1600);
  }

  document.addEventListener('click', function(ev){
    var t = ev.target.closest('[data-act]');
    if (!t) return;
    var act = t.getAttribute('data-act');
    if (act === 'sort') {
      sortMode = t.getAttribute('data-mode');
      var c = compute();
      document.getElementById('board').innerHTML = boardHtml(c);
      document.querySelectorAll('[data-act="sort"]').forEach(function(x){ x.setAttribute('aria-pressed', String(x.getAttribute('data-mode') === sortMode)); });
    } else if (act === 'copy') copyHow(t);
  });

  render();

})();

</script>
